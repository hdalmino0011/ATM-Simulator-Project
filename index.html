class ATMTransactionError(Exception):
    """Base custom exception for all invalid ATM inputs and business rule violations."""
    def __init__(self, message: str, error_code: str = "TRANSACTION_ERROR"):
        super().__init__(message)
        self.message = message
        self.error_code = error_code

class InvalidAmountError(ATMTransactionError):
    pass

class InsufficientFundsError(ATMTransactionError):
    pass

class WithdrawalLimitError(ATMTransactionError):
    pass

class MaintainingBalanceError(ATMTransactionError):
    pass

class Account:
    """Parent class encapsulating _account_number, _holder_name, _balance, and _transactions."""
    def __init__(self, account_number: str, holder_name: str, starting_balance: float, account_type: str = "Standard Account"):
        self._account_number = account_number
        self._holder_name = holder_name
        self._balance = round(float(starting_balance), 2)
        self._account_type = account_type
        self._transactions = []

    @property
    def balance(self) -> float:
        return self._balance

    def deposit(self, raw_amount, note="Cash Deposit via Express ATM"):
        amount = self.parse_positive_amount(raw_amount, "Deposit")
        self._balance = round(self._balance + amount, 2)
        return self._record_transaction("Deposit", amount, 0.0, note)

    def withdraw(self, raw_amount, note="Cash Withdrawal"):
        """Overridden by SavingsAccount and CheckingAccount (Polymorphism)."""
        raise NotImplementedError("Subclasses must override withdraw()")

class SavingsAccount(Account):
    """Child Class 1: Requires PHP 500.00 maintaining balance, max PHP 20,000.00 per withdrawal, PHP 0.00 fee."""
    MAINTAINING_BALANCE = 500.00
    WITHDRAWAL_LIMIT = 20000.00
    WITHDRAWAL_FEE = 0.00

    def withdraw(self, raw_amount, note="Savings ATM Withdrawal"):
        amount = self.parse_positive_amount(raw_amount, "Savings Withdrawal")
        if amount > self.WITHDRAWAL_LIMIT:
            raise WithdrawalLimitError("Savings withdrawal cannot exceed PHP 20,000.00 per transaction.")
        if amount > self._balance:
            raise InsufficientFundsError("Insufficient funds in Savings Account.")
        if (self._balance - amount) < self.MAINTAINING_BALANCE:
            raise MaintainingBalanceError("Cannot drop below PHP 500.00 maintaining balance.")
        self._balance = round(self._balance - amount, 2)
        return self._record_transaction("Withdrawal", amount, 0.0, note)

class CheckingAccount(Account):
    """Child Class 2: PHP 0.00 maintaining balance, max PHP 50,000.00 per withdrawal, flat PHP 15.00 fee."""
    MAINTAINING_BALANCE = 0.00
    WITHDRAWAL_LIMIT = 50000.00
    WITHDRAWAL_FEE = 15.00

    def withdraw(self, raw_amount, note="Checking ATM Withdrawal"):
        amount = self.parse_positive_amount(raw_amount, "Checking Withdrawal")
        if amount > self.WITHDRAWAL_LIMIT:
            raise WithdrawalLimitError("Checking withdrawal cannot exceed PHP 50,000.00 per transaction.")
        total_deduction = round(amount + self.WITHDRAWAL_FEE, 2)
        if total_deduction > self._balance:
            raise InsufficientFundsError("Insufficient funds to cover withdrawal plus PHP 15.00 checking fee.")
        self._balance = round(self._balance - total_deduction, 2)
        return self._record_transaction("Withdrawal", amount, self.WITHDRAWAL_FEE, note)
